\section{Raw segmentation performance}

    columns/CannyFill/.style={column name=CannyFill},
    columns/GCautobrush/.style={column name=GCautobrush},
    columns/Gaussian/.style={column name=Gaussian},
    columns/Otsu/.style={column name=Otsu},
    columns/mSAM/.style={column name=mSAM},
}


\begin{table}[ht]
\centering
\caption{Segmentation \acs{Dice} score on easy images.}
\subcaption{Label column represents the name of the document segmented, while all other columns represent \acs{Dice} scores obtained for those documents for a given model.}
\pgfplotstabletypeset[
    col sep=comma,
    fixed,
    precision=3,
    columns={label,\modelone,\modeltwo,\modelthree,\modelfour,\modelfive},
    columns/label/.style={
        string type,
        column name=Label
    },
    row predicate/.code={
        \pgfplotstablegetelem{\pgfplotstablerow}{difficulty}\of{\rawpivot}
        \edef\temp{\pgfplotsretval}
        \def\match{easy}
        \ifx\temp\match
        \else
            \pgfplotstableuserowfalse
        \fi
    },
    every head row/.style={before row=\toprule,after row=\midrule},
    every last row/.style={after row=\bottomrule}
]{\rawpivot}
\end{table}
