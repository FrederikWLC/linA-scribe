\begin{table}[ht]
\centering
\caption{Raw \acs{Dice} score data on easy images.
Label column represents the name of the document segmented, while all other columns represent \acs{Dice} scores obtained for those documents for a given model.}
\pgfplotstabletypeset[
    col sep=comma,
    fixed,
    precision=3,
    columns={label,\modelone,\modeltwo,\modelthree,\modelfour,\modelfive,\modelsix},
    columns/label/.style={
        string type,
        column name=Label
    },
    row predicate/.code={
        \pgfplotstablegetelem{\pgfplotstablerow}{difficulty}\of{\rawpivot}
        \edef\temp{\pgfplotsretval}
        \def\match{easy}
        \ifx\temp\match
        \else
            \pgfplotstableuserowfalse
        \fi
    },
    every head row/.style={before row=\toprule,after row=\midrule},
    every last row/.style={after row=\bottomrule}
]{\rawpivot}
\end{table}
